<script>
  const START_DATE = new Date('2026-05-19');
  const TOTAL_DAYS = 270;

  const MILESTONES = [
    { id: 'foia10', text: '10 FOIA requests filed', icon: '📁' },
    { id: 'foia100', text: '100 FOIA requests filed', icon: '📁' },
    { id: 'firstupload', text: 'First FAF video uploaded', icon: '🎬' },
    { id: 'sub100', text: '100 subscribers', icon: '👥' },
    { id: 'sub500', text: '500 subscribers', icon: '👥' },
    { id: 'sub1k', text: '1,000 subscribers', icon: '👥' },
    { id: 'hours4k', text: '4,000 watch hours hit', icon: '⏱️' },
    { id: 'adsense', text: 'AdSense application submitted', icon: '💳' },
    { id: 'first100', text: 'First $100 earned', icon: '💵' },
    { id: 'first1k', text: 'First $1,000 month', icon: '💰' },
    { id: 'first5k', text: 'First $5,000 month', icon: '💰' },
    { id: 'llc', text: 'Nevada LLC formed', icon: '📋' },
    { id: 'saved20k', text: '$20,000 saved', icon: '🏦' },
    { id: 'move', text: 'Moved to Henderson, NV', icon: '🏠' },
  ];

  const DAILY_TASKS = [
    { id: 'foia', text: '10 FOIA requests filed' },
    { id: 'trading', text: 'Swing trading study (1–2 hrs + notes)' },
    { id: 'faf', text: 'FAF editing/production work' },
    { id: 'analytics', text: 'Analytics & content optimization' },
    { id: 'nowork', text: 'Relaxed ONLY after all work done' },
  ];

  // STATE
  let state = {
    days: {},        // { 1: 'complete' | 'failed' | null }
    income: 0,
    milestones: {},  // { id: true }
    todayTasks: {},  // { taskId: true } — resets each calendar day
    todayTasksDate: null,
  };

  function save() { localStorage.setItem('grind270', JSON.stringify(state)); }

  function load() {
    const raw = localStorage.getItem('grind270');
    if (raw) state = { ...state, ...JSON.parse(raw) };
  }

  function getCurrentDay() {
    const now = new Date();
    const diff = Math.floor((now - START_DATE) / 86400000) + 1;
    return Math.max(1, Math.min(diff, TOTAL_DAYS));
  }

  function formatMoney(n) {
    if (n >= 1000000) return '$' + (n/1000000).toFixed(1) + 'M';
    if (n >= 1000) return '$' + (n/1000).toFixed(1) + 'K';
    return '$' + n.toLocaleString();
  }

  function showToast(msg) {
    const t = document.getElementById('toast');
    t.textContent = msg;
    t.classList.add('show');
    setTimeout(() => t.classList.remove('show'), 2200);
  }

  function showView(name) {
    document.querySelectorAll('.view').forEach(v => v.classList.remove('active'));
    document.querySelectorAll('.nav-btn').forEach(b => b.classList.remove('active'));
    document.getElementById('view-' + name).classList.add('active');
    document.getElementById('nav-' + name).classList.add('active');
  }

  function addIncome() {
    const val = parseFloat(document.getElementById('income-input').value);
    if (isNaN(val) || val <= 0) return;
    state.income += val;
    document.getElementById('income-input').value = '';
    save();
    render();
    showToast('+ ' + formatMoney(val) + ' added 💰');
  }

  function setIncome() {
    const val = parseFloat(document.getElementById('income-input').value);
    if (isNaN(val) || val < 0) return;
    state.income = val;
    document.getElementById('income-input').value = '';
    save();
    render();
    showToast('Income set to ' + formatMoney(val));
  }

  function toggleDay(dayNum) {
    const cur = state.days[dayNum];
    if (cur === 'complete') state.days[dayNum] = 'failed';
    else if (cur === 'failed') state.days[dayNum] = null;
    else state.days[dayNum] = 'complete';
    save();
    render();
  }

  function markToday() {
    const day = getCurrentDay();
    state.days[day] = 'complete';
    // reset today tasks
    state.todayTasks = {};
    Object.keys(state.todayTasks);
    save();
    render();
    showToast('Day ' + day + ' LOCKED IN ✅');
  }

  function toggleTask(taskId) {
    const today = new Date().toDateString();
    if (state.todayTasksDate !== today) {
      state.todayTasks = {};
      state.todayTasksDate = today;
    }
    state.todayTasks[taskId] = !state.todayTasks[taskId];
    save();
    render();
  }

  function toggleMilestone(id) {
    state.milestones[id] = !state.milestones[id];
    save();
    render();
    if (state.milestones[id]) showToast('Milestone unlocked! 🎯');
  }

  function calcStreak() {
    let streak = 0;
    const day = getCurrentDay();
    for (let i = day; i >= 1; i--) {
      if (state.days[i] === 'complete') streak++;
      else break;
    }
    return streak;
  }

  function render() {
    const day = getCurrentDay();
    const completed = Object.values(state.days).filter(v => v === 'complete').length;
    const streak = calcStreak();
    const pct = Math.round((completed / TOTAL_DAYS) * 100);

    // Stats
    document.getElementById('current-day-stat').textContent = day;
    document.getElementById('days-left-stat').textContent = Math.max(0, TOTAL_DAYS - day + 1);
    document.getElementById('streak-stat').textContent = streak;
    document.getElementById('completed-stat').textContent = completed;
    document.getElementById('progress-pct').textContent = pct + '%';
    document.getElementById('progress-fill').style.width = pct + '%';
    document.getElementById('income-display').textContent = formatMoney(state.income);

    // Today header
    document.getElementById('today-day-header').textContent = 'DAY ' + day;
    const phase = day <= 90 ? 'GRIND PHASE · Days 1–90 · 10 hrs/day' :
                  day <= 180 ? 'GROWTH PHASE · Days 91–180 · 8 hrs/day' :
                  'SCALE PHASE · Days 181–270 · 8 hrs/day';
    document.getElementById('today-phase-label').textContent = phase;

    // Today tasks
    const today = new Date().toDateString();
    if (state.todayTasksDate !== today) state.todayTasks = {};
    const cl = document.getElementById('daily-checklist');
    cl.innerHTML = DAILY_TASKS.map(t => `
      <div class="checklist-item ${state.todayTasks[t.id] ? 'checked' : ''}" onclick="toggleTask('${t.id}')">
        <div class="check-box">${state.todayTasks[t.id] ? '✓' : ''}</div>
        <div class="check-label">${t.text}</div>
      </div>
    `).join('');

    const allDone = DAILY_TASKS.every(t => state.todayTasks[t.id]);
    const btn = document.getElementById('mark-day-btn');
    const todayDone = state.days[day] === 'complete';
    btn.textContent = todayDone ? '✅ DAY ' + day + ' COMPLETE' : '✅ MARK DAY COMPLETE';
    btn.style.opacity = todayDone ? '0.5' : '1';

    // Milestones
    document.getElementById('milestones-list').innerHTML = MILESTONES.map(m => `
      <div class="milestone ${state.milestones[m.id] ? 'done' : ''}" onclick="toggleMilestone('${m.id}')">
        <div class="milestone-check">${state.milestones[m.id] ? '✓' : ''}</div>
        <div class="milestone-text">${m.icon} ${m.text}</div>
      </div>
    `).join('');

    // Day grids
    ['phase1','phase2','phase3'].forEach((phase, pi) => {
      const start = pi * 90 + 1;
      const end = start + 89;
      let html = '';
      for (let d = start; d <= end; d++) {
        const status = state.days[d];
        const isToday = d === day;
        const cls = [
          'day-dot',
          status === 'complete' ? 'complete' : '',
          status === 'failed' ? 'failed' : '',
          isToday ? 'today' : ''
        ].filter(Boolean).join(' ');
        const emoji = status === 'complete' ? '✓' : status === 'failed' ? '✗' : d;
        html += `<div class="${cls}" onclick="toggleDay(${d})" title="Day ${d}">${d <= day ? (status === 'complete' ? '✓' : status === 'failed' ? '✗' : d) : ''}</div>`;
      }
      document.getElementById('grid-' + phase).innerHTML = html;
    });
  }
